\section{Recorridos clásicos}

\subsection{DFS (Depth-First Search)}

\begin{lstlisting}[caption=DFS no dirigido]
	bool vis[MAXN];
	vector<int> G[MAXN];
	
	void dfs(int u) {
		vis[u] = true;
		for (int v : G[u]) {
			if (!vis[v]) dfs(v);
		}
	}
\end{lstlisting}

\subsection{BFS (Breadth-First Search)}

BFS recorre el grafo por capas (nivel por nivel), lo que garantiza que la primera vez que llega a un nodo, lo hace por el \textbf{camino más corto} (en cantidad de aristas).

\subsubsection*{BFS básico con distancias}

\begin{lstlisting}[caption=BFS con calculo de distancias]
	int dist[MAXN];
	bool vis[MAXN];
	vector<int> G[MAXN];
	
	void bfs(int s) {
		queue<int> q;
		q.push(s);
		vis[s] = true;
		dist[s] = 0;
		
		while (!q.empty()) {
			int u = q.front(); q.pop();
			for (int v : G[u]) {
				if (!vis[v]) {
					vis[v] = true;
					dist[v] = dist[u] + 1;
					q.push(v);
				}
			}
		}
	}
\end{lstlisting}

Despues de llamar \texttt{bfs(s)}, \texttt{dist[v]} contiene la distancia minima (en cantidad de aristas) desde \texttt{s} hasta \texttt{v}. Si \texttt{vis[v] == false}, el nodo \texttt{v} no es alcanzable desde \texttt{s}.

\subsubsection*{Obtener la distancia mínima entre dos nodos}

Para saber la distancia entre \texttt{s} y un nodo destino \texttt{t}:

\begin{lstlisting}[caption=Consultar distancia minima]
	bfs(s);
	
	if (!vis[t]) {
		// t no es alcanzable desde s
	} else {
		cout << dist[t] << "\n"; // distancia minima s -> t
	}
\end{lstlisting}

\subsubsection*{Reconstruir el camino mínimo}

Para recuperar \textbf{la secuencia de nodos} del camino mas corto, se agrega un arreglo \texttt{parent[]} que registra, para cada nodo, desde cual nodo fue visitado por primera vez.

\bigskip

\textbf{BFS con arreglo parent:}

\begin{lstlisting}[caption=BFS con reconstruccion de camino]
	int dist[MAXN], par[MAXN];
	bool vis[MAXN];
	vector<int> G[MAXN];
	
	void bfs(int s, int n) {
		fill(vis, vis + n, false);
		fill(par, par + n, -1);
		
		queue<int> q;
		vis[s] = true;
		dist[s] = 0;
		q.push(s);
		
		while (!q.empty()) {
			int u = q.front(); q.pop();
			for (int v : G[u]) {
				if (!vis[v]) {
					vis[v] = true;
					dist[v] = dist[u] + 1;
					par[v] = u;  // u es el padre de v
					q.push(v);
				}
			}
		}
	}
\end{lstlisting}

\textbf{Reconstruir el camino desde \texttt{s} hasta \texttt{t}:}

Una vez que termina el BFS, el camino esta encadenado hacia atras a traves de \texttt{par[]}. Se sigue desde \texttt{t} hasta llegar a \texttt{-1} (el origen), y luego se invierte.

\begin{lstlisting}[caption=Funcion para reconstruir el camino]
	vector<int> getPath(int s, int t) {
		if (!vis[t]) return {}; // t no alcanzable
		
		vector<int> path;
		for (int v = t; v != -1; v = par[v])
			path.push_back(v);
		
		reverse(path.begin(), path.end());
		// path = {s, ..., t}
		return path;
	}
\end{lstlisting}

\textbf{Ejemplo completo:}

\begin{lstlisting}[caption=Ejemplo s=0 a t=4]
	bfs(0, n);
	
	auto path = getPath(0, 4);
	
	cout << "Distancia: " << dist[4] << "\n";
	cout << "Camino: ";
	for (int v : path) cout << v << " ";
	cout << "\n";
	// Salida: Distancia: 3
	//         Camino: 0 2 3 4
\end{lstlisting}

\begin{itemize}
    \item \textbf{Distancia}: \texttt{dist[t]} despues de correr \texttt{bfs(s)}.
    \item \textbf{Camino}: seguir \texttt{par[t] -> par[par[t]] -> ...} hasta llegar a \texttt{-1}, luego invertir.
    \item \textbf{Complejidad}: $O(n + m)$ para construccion del BFS.
\end{itemize}


\section{Componentes conexas}

Para contar la cantidad de componentes conexas en un grafo no dirigido y guardar un nodo representativo (representante) de cada una de ellas, podemos usar el siguiente enfoque con DFS:

\begin{lstlisting}[language=C++, caption={Contar componentes conexas y guardar representantes}]
const ll MAXN = 1e5 + 10;
bool vis[MAXN];
vector<ll> g[MAXN];
vector<ll> lista; // Guarda un nodo representativo de cada componente

void dfs(ll u) {
    vis[u] = true;
    for (ll v : g[u]) {
        if (!vis[v]) {
            dfs(v);
        }
    }
}

ll comp(ll n) {
    ll ans = 0;
    lista.clear();
    fore(i, 0, n) {
        if (!vis[i]) {
            dfs(i);
            ans++;
            lista.pb(i + 1); // Guarda el nodo representativo (1-indexed)
        }
    }
    return ans;
}
\end{lstlisting}

\subsection*{Detalles de uso}
\begin{itemize}
    \item \textbf{Resultado:} Al llamar a \texttt{comp(n)}, la función devuelve la **cantidad total de componentes conexas** del grafo.
    \item \textbf{Representantes:} El vector \texttt{lista} contendrá un nodo representativo por cada componente hallada (en este caso guardado con indexación 1, \texttt{i + 1}).
    \item \textbf{Complejidad:} $\mathcal{O}(n + m)$ ya que se visita cada nodo y cada arista una sola vez.
\end{itemize}
\section{Dijkstra (Caminos Mínimos con Pesos Positivos)}

El algoritmo de Dijkstra permite encontrar la distancia mínima desde un nodo origen $s$ a todos los demás nodos en un grafo con \textbf{pesos no negativos}. Utiliza una cola de prioridad (\texttt{priority\_queue}) para procesar siempre el nodo con la menor distancia acumulada hasta el momento.

\subsection*{Representación y Lectura de un Grafo con Pesos}

Para almacenar un grafo con pesos, la lista de adyacencia guarda pares de la forma \texttt{\{vecino, peso\}}:

\begin{lstlisting}[language=C++, caption={Representación de grafo ponderado}]
const ll MAXN = 1e5 + 10;
// g[u] contiene pares {v, w} donde v es el destino y w el peso de la arista
vector<pair<ll, ll>> g[MAXN]; 

void agregar_arista(ll u, ll v, ll w, bool dirigido = false) {
    g[u].push_back({v, w});
    if (!dirigido) {
        g[v].push_back({u, w}); // Si el grafo es no dirigido
    }
}
\end{lstlisting}

Para leer las aristas desde la entrada estándar ($1$-indexed):

\begin{lstlisting}[language=C++, caption={Lectura de aristas ponderadas}]
ll n, m; cin >> n >> m;
fore(i, 0, m) {
    ll u, v, w; 
    cin >> u >> v >> w;
    g[u].push_back({v, w});
    // g[v].push_back({u, w}); // Descomentar si es no dirigido
}
\end{lstlisting}

\subsection*{Implementación de Dijkstra}

La \texttt{priority\_queue} en C++ ordena los elementos de mayor a menor por defecto. Para que funcione como una \textit{min-heap} (menor distancia primero), se utiliza \texttt{greater<pair<ll,ll>>}.

\begin{lstlisting}[language=C++, caption={Algoritmo de Dijkstra con C++}]
const ll INF = 1e18 + 10;
const ll MAXN = 1e5 + 10;

vector<pair<ll, ll>> g[MAXN]; // {destino, peso}
ll dist[MAXN];
ll par[MAXN]; // Para reconstruccion del camino (opcional)

void dijkstra(ll s, ll n) {
    // priority_queue guarda pares {distancia_acumulada, nodo}
    priority_queue<pair<ll, ll>, vector<pair<ll, ll>>, greater<pair<ll, ll>>> pq;
    
    fill(dist, dist + n + 1, INF);
    fill(par, par + n + 1, -1);
    
    dist[s] = 0;
    pq.push({0, s});
    
    while (!pq.empty()) {
        auto [d, u] = pq.top(); 
        pq.pop();
        
        // Si encontramos una distancia guardada menor, ignoramos esta entrada desactualizada
        if (d > dist[u]) continue;
        
        for (auto [v, w] : g[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                par[v] = u; // Guardar el padre para reconstruir el camino
                pq.push({dist[v], v});
            }
        }
    }
}
\end{lstlisting}

\subsection*{Reconstrucción del Camino Mínimo}

Para obtener la secuencia exacta de nodos del camino mínimo desde $s$ hasta $t$:

\begin{lstlisting}[language=C++, caption={Recuperar el camino desde origen s a destino t}]
vector<ll> obtenerCamino(ll s, ll t) {
    if (dist[t] == INF) return {}; // t no es alcanzable desde s
    
    vector<ll> camino;
    for (ll v = t; v != -1; v = par[v]) {
        camino.push_back(v);
    }
    reverse(camino.begin(), camino.end());
    return camino;
}
\end{lstlisting}

\subsection*{Detalles de uso y Complejidad}
\begin{itemize}
    \item \textbf{Complejidad temporal:} $\mathcal{O}((N + M) \log N)$, adecuada para $N, M \le 10^5$ o $2 \cdot 10^5$.
    \item \textbf{Tipo de datos:} Se utiliza \texttt{ll} (\texttt{long long}) y \texttt{INF = 1e18} para prevenir desbordamientos (\textit{overflow}) durante las sumas de pesos.
    \item \textbf{Restricción:} Solo funciona si todos los pesos $w \ge 0$. Si existen pesos negativos, usar Bellman-Ford o SPFA.
\end{itemize}

\section{Bellman-Ford}

\begin{lstlisting}[caption=Bellman-Ford para grafos con pesos negativos]
	vector<tuple<int, int, int>> edges; // {u, v, peso}
	int dist[MAXN];
	
	bool bellman(int s, int n) {
		fill(dist, dist + MAXN, INF);
		dist[s] = 0;
		
		for (int i = 0; i < n - 1; ++i) {
			for (auto [u, v, w] : edges) {
				if (dist[u] + w < dist[v]) {
					dist[v] = dist[u] + w;
				}
			}
		}
		
		// Detectar ciclos negativos
		for (auto [u, v, w] : edges) {
			if (dist[u] + w < dist[v]) return true;
		}
		
		return false;
	}
\end{lstlisting}

\section{Problemas clásicos}

\subsection*{A - Número de componentes conexas}

\textbf{Estado:} nodos visitados.  
\textbf{Recorrido:} DFS o BFS.  
\textbf{Complejidad:} \( O(n + m) \)

\subsection*{B - Camino más corto (no ponderado)}

\textbf{Idea:} usar BFS desde el nodo origen y guardar las distancias.

\subsection*{C - Dijkstra / Bellman-Ford}

\textbf{Cuando usar cada uno:}

- Dijkstra → aristas no negativas.
- Bellman-Ford → permite pesos negativos.

\section{¿Cómo verificar si un grafo es una línea (path)?}



\begin{enumerate}
	\item Verificar que \( M = N - 1 \) (condición necesaria para ser un árbol).
	\item Verificar que el grafo es conexo (DFS o BFS desde cualquier nodo).
	\item Contar los grados de los nodos:
	\begin{itemize}
		\item Debe haber exactamente 2 nodos con grado 1.
		\item Todos los demás deben tener grado 2.
	\end{itemize}
\end{enumerate}

\subsection*{Código en C++}

\begin{lstlisting}[caption=Verificar si un grafo es una línea]
	vector<int> G[MAXN];
	bool vis[MAXN];
	int deg[MAXN];
	
	void dfs(int u) {
		vis[u] = true;
		for (int v : G[u]) {
			if (!vis[v]) dfs(v);
		}
	}
	
	bool es_linea(int n, int m) {
		if (m != n - 1) return false; // debe tener N-1 aristas
		
		// verificar conectividad
		fill(vis, vis + n, false);
		dfs(0);
		for (int i = 0; i < n; ++i)
		if (!vis[i]) return false;
		
		// contar grados
		int cnt_deg1 = 0;
		for (int i = 0; i < n; ++i) {
			if (deg[i] == 1) cnt_deg1++;
			else if (deg[i] != 2) return false;
		}
		
		return cnt_deg1 == 2;
	}
\end{lstlisting}

\textbf{Entrada:}
\begin{itemize}
	\item Llenar \texttt{G[i]} con las aristas del grafo.
	\item Aumentar \texttt{deg[u]} y \texttt{deg[v]} cada vez que se agrega una arista entre \( u \) y \( v \).
\end{itemize}
		
\section{Detección de Ciclos}

\subsection{Grafo No Dirigido}

En un grafo no dirigido, un ciclo existe si durante un recorrido DFS alcanzamos un nodo ya visitado que \textbf{no es el padre} directo del nodo actual.

\subsubsection*{Algoritmo y Reconstrucción}
Para recuperar los nodos que forman el ciclo:
\begin{enumerate}
    \item Se guardan los extremos de la arista que cierra el ciclo en las variables \texttt{cicloU} y \texttt{cicloV}.
    \item Se reconstruye el camino desde \texttt{cicloU} hacia atrás usando \texttt{par[]} hasta llegar a \texttt{cicloV}.
    \item Se añade \texttt{cicloV} al final para cerrar el ciclo explícitamente.
\end{enumerate}

\begin{lstlisting}[language=C++, caption={Detectar y reconstruir ciclo en grafo no dirigido (1-indexed)}]
const ll MAXN = 1e5 + 10;
bool vis[MAXN];
ll par[MAXN];
vector<ll> g[MAXN];
ll cicloU = -1, cicloV = -1;

bool dfs(ll u, ll parent) {
    vis[u] = true;
    par[u] = parent;
    
    for (ll v : g[u]) {
        if (!vis[v]) {
            if (dfs(v, u)) return true;
        } else if (v != parent) {
            // Se encontró una arista de retroceso que cierra el ciclo
            cicloU = u;
            cicloV = v;
            return true;
        }
    }
    return false;
}

vector<ll> obtenerCiclo() {
    vector<ll> ciclo;
    // Reconstruimos el camino desde cicloU hasta cicloV a traves de los padres
    for (ll curr = cicloU; curr != cicloV; curr = par[curr]) {
        ciclo.push_back(curr);
    }
    ciclo.push_back(cicloV);
    ciclo.push_back(cicloU); // Cerrar el ciclo (opcional)
    return ciclo;
}
\end{lstlisting}

\subsubsection*{Cómo usarlo}

\begin{lstlisting}[language=C++, caption={Uso en la funcion principal}]
int main() {
    // Lectura e inicializacion de variables...
    fill(vis, vis + n + 1, false);
    fill(par, par + n + 1, -1);
    cicloU = -1; cicloV = -1;

    bool hay_ciclo = false;
    fore(i, 1, n + 1) { // Iteracion 1-indexed para todas las componentes
        if (!vis[i]) {
            if (dfs(i, -1)) {
                hay_ciclo = true;
                break;
            }
        }
    }

    if (hay_ciclo) {
        vector<ll> ciclo = obtenerCiclo();
        cout << "Ciclo detectado de tamano " << ciclo.size() - 1 << ":\n";
        for (ll nodo : ciclo) cout << nodo << " ";
        cout << "\n";
    } else {
        cout << "IMPOSSIBLE\n";
    }
}
\end{lstlisting}

\subsection{Grafo Dirigido}

En un grafo dirigido se utiliza el algoritmo de colores para identificar **aristas de retroceso** (\textit{back edges}), las cuales apuntan a nodos que aún se encuentran en la pila de recursión activa.

\begin{itemize}
    \item \textbf{0 (Blanco):} Nodo no visitado.
    \item \textbf{1 (Gris):} Nodo en proceso de exploración (en la pila de recursión).
    \item \textbf{2 (Negro):} Nodo completamente visitado.
\end{itemize}

\begin{lstlisting}[language=C++, caption={Detectar ciclo en grafo dirigido}]
int color[MAXN]; // 0: blanco, 1: gris, 2: negro
vector<ll> g[MAXN];

bool dfsDirigido(ll u) {
    color[u] = 1; // En proceso (gris)
    
    for (ll v : g[u]) {
        if (color[v] == 0) {
            if (dfsDirigido(v)) return true;
        } else if (color[v] == 1) {
            return true; // Encontrado nodo en la pila -> Hay ciclo
        }
    }
    
    color[u] = 2; // Procesado (negro)
    return false;
}

bool hayCicloDirigido(ll n) {
    fill(color, color + n + 1, 0);
    fore(i, 1, n + 1) {
        if (color[i] == 0 && dfsDirigido(i))
            return true;
    }
    return false;
}
\end{lstlisting}

\section{Verificar si hay ciclo impar}

\begin{enumerate}
	\item Intentar colorear el grafo con 2 colores (bipartición)
	\item Si no es posible, contiene ciclo impar
\end{enumerate}

\subsection*{Código en C++}

\begin{lstlisting}[caption=Detectar ciclo impar]
	vector<int> G[MAXN];
	int col[MAXN]; // 0 o 1
	
	bool tieneCicloImpar(int n) {
		queue<int> q;
		fill(col, col + n, -1);
		
		col[0] = 0;
		q.push(0);
		
		while (!q.empty()) {
			int u = q.front(); q.pop();
			for (int v : G[u]) {
				if (col[v] == -1) {
					col[v] = 1 - col[u];
					q.push(v);
				}
				else if (col[v] == col[u]) {
					return true; // Ciclo impar encontrado
				}
			}
		}
		return false;
	}
\end{lstlisting}

\section{Verificar si el grafo es bipartito (o tiene ciclo impar)}

Un grafo es \textbf{bipartito} si sus vértices se pueden dividir en dos conjuntos tales que no existan aristas entre vértices del mismo conjunto. Esto es equivalente a decir que el grafo **no contiene ciclos impares**.

\subsection*{Algoritmo}
\begin{enumerate}
    \item Se intenta colorear el grafo con 2 colores (\texttt{0} y \texttt{1}) usando BFS.
    \item Si al recorrer un vecino este ya fue coloreado y tiene el mismo color que el nodo actual (\texttt{col[v] == col[u]}), significa que existe un ciclo impar y el grafo \textbf{no es bipartito}.
    \item Se debe iterar sobre todos los nodos del grafo para asegurar la verificación en **todas las componentes conexas**.
\end{enumerate}

\subsection*{Código en C++}

\begin{lstlisting}[language=C++, caption={Verificar si un grafo es bipartito (1-indexed)}]
const ll MAXN = 1e5 + 10;
ll col[MAXN];
vector<ll> g[MAXN];

bool bfs(ll s) {
    queue<ll> q;
    col[s] = 0;
    q.push(s);
    
    while (!q.empty()) {
        ll u = q.front();
        q.pop();
        for (ll v : g[u]) {
            if (col[v] == -1) {
                col[v] = 1 - col[u]; // Color opuesto
                q.push(v);
            } else if (col[v] == col[u]) {
                return false; // Conflicto de colores (ciclo impar)
            }
        }
    }
    return true;
}

bool esBipar(ll n) {
    fill(col, col + n + 1, -1);
    fore(i, 1, n + 1) { // Iterar sobre todas las componentes (1-indexed)
        if (col[i] == -1) {
            if (!bfs(i)) return false;
        }
    }
    return true;
}
\end{lstlisting}

\subsection*{Detalles de uso}
\begin{itemize}
    \item \textbf{Indexación:} El código asume nodos numerados de $1$ a $n$. Si es $0$-indexed, ajustar la iteración en \texttt{esBipar} a \texttt{fore(i, 0, n)}.
    \item \textbf{Complejidad:} $\mathcal{O}(n + m)$ ya que procesa cada nodo y arista una sola vez.
\end{itemize}

\section{Verificar si el grafo es un árbol}

\begin{enumerate}
	\item Verificar que tiene exactamente N-1 aristas
	\item Verificar que es conexo
	\item Verificar que no tiene ciclos
\end{enumerate}

\subsection*{Código en C++}

\begin{lstlisting}[caption=Verificar si grafo es árbol]
	vector<int> G[MAXN];
	bool vis[MAXN];
	
	bool hasCycle(int u, int parent) {
		vis[u] = true;
		for (int v : G[u]) {
			if (!vis[v]) {
				if (hasCycle(v, u)) return true;
			}
			else if (v != parent) return true;
		}
		return false;
	}
	
	bool esArbol(int n, int m) {
		if (m != n - 1) return false;
		
		// Verificar conexidad
		fill(vis, vis + n, false);
		hasCycle(0, -1);
		for (int i = 0; i < n; ++i)
		if (!vis[i]) return false;
		
		// Verificar que no tiene ciclos
		fill(vis, vis + n, false);
		if (hasCycle(0, -1)) return false;
		
		return true;
	}
\end{lstlisting}

\section{Operaciones básicas en árboles}

\subsection*{Obtener el padre de cada nodo}

Durante un DFS, el nodo desde el que llegaste es el padre.

\begin{lstlisting}[caption=Obtener padre de cada nodo]
	vector<vector<ll>> arb(n);
	vector<ll> padre(n, -1);
	
	void dfs(ll u, ll p){
		padre[u] = p;
		
		for(ll v : arb[u]){
			if(v == p) continue;
			dfs(v, u);
		}
	}
\end{lstlisting}

Llamás:

\begin{lstlisting}[caption=Llamada inicial del DFS]
	dfs(0, -1);
\end{lstlisting}

\subsection*{Obtener la cantidad de hijos}

Cada vez que desde un nodo visitás un vecino que no es su padre, encontraste un hijo.

\begin{lstlisting}[caption=Obtener cantidad de hijos]
	vector<ll> hijos(n, 0);
	
	void dfs(ll u, ll p){
		padre[u] = p;
		
		for(ll v : arb[u]){
			if(v == p) continue;
			
			hijos[u]++;
			
			dfs(v, u);
		}
	}
\end{lstlisting}

\subsection*{Obtener el grado de cada nodo}

El grado no depende de la raíz. Es simplemente la cantidad de vecinos.

Como estás usando listas de adyacencia:

\begin{lstlisting}[caption=Obtener grado de un nodo]
	vector<ll> grado(n);
	
	for(ll i = 0; i < n; i++)
		grado[i] = arb[i].size();
\end{lstlisting}

O directamente imprimiendo:

\section{Diámetro de un árbol}

El algoritmo más usado en programación competitiva para obtener el diámetro de un árbol es hacer dos DFS (o dos BFS).

\subsection*{Idea}

El diámetro es el camino simple más largo del árbol. En un árbol, si empezás desde cualquier nodo y buscás el más lejano, siempre terminás en uno de los extremos del diámetro. Luego, al volver a buscar desde ese extremo, llegás al otro extremo y la distancia recorrida es exactamente el diámetro.

\subsection*{Algoritmo}

\begin{enumerate}
	\item Elegí un nodo cualquiera (por ejemplo, el 0 o el 1).
	\item Hacé un DFS/BFS desde ese nodo y encontrá el vértice más lejano $u$.
	\item Hacé otro DFS/BFS desde $u$ y encontrá el vértice más lejano $v$.
	\item La distancia entre $u$ y $v$ es el diámetro.
\end{enumerate}

\textbf{Complejidad:} $\mathcal{O}(n)$.

\subsection*{Código en C++}

\begin{lstlisting}[caption=Obtener diámetro de un árbol]
	#include <bits/stdc++.h>
	using namespace std;
	
	using ll = long long;
	
	vector<vector<int>> g;
	int mxDist, farthest;
	
	void dfs(int v, int p, int d){
		if(d > mxDist){
			mxDist = d;
			farthest = v;
		}
		
		for(int u : g[v]){
			if(u != p)
				dfs(u, v, d + 1);
		}
	}
	
	int main(){
		int n;
		cin >> n;
		
		g.assign(n, {});
		
		for(int i = 0; i < n - 1; i++){
			int u, v;
			cin >> u >> v;
			--u; --v;
			
			g[u].push_back(v);
			g[v].push_back(u);
		}
		
		mxDist = -1;
		dfs(0, -1, 0);
		int extremo1 = farthest;
		
		mxDist = -1;
		dfs(extremo1, -1, 0);
		int extremo2 = farthest;
		
		cout << "Diametro = " << mxDist << "\n";
		cout << "Extremos: " << extremo1 + 1 << " " << extremo2 + 1 << "\n";
	}
\end{lstlisting}

\subsection*{Recuperar el camino del diámetro}

Si además querés recuperar el camino del diámetro, guardá el padre en el segundo DFS.

\begin{lstlisting}[caption=DFS guardando padres]
	vector<int> parent;
	
	void dfs(int v, int p, int d){
		parent[v] = p;
		
		if(d > mxDist){
			mxDist = d;
			farthest = v;
		}
		
		for(int u : g[v]){
			if(u != p)
				dfs(u, v, d + 1);
		}
	}
\end{lstlisting}

Después del segundo DFS:

\begin{lstlisting}[caption=Reconstruir camino del diámetro]
	vector<int> path;
	for(int x = extremo2; x != -1; x = parent[x])
		path.push_back(x);
	
	reverse(path.begin(), path.end());
\end{lstlisting}

Ahora \texttt{path} contiene todos los nodos del diámetro en orden.

\subsection*{Variante con pesos}

Si las aristas tienen pesos positivos, el algoritmo es exactamente el mismo, pero en lugar de sumar 1 en el DFS hacés:

\begin{lstlisting}[caption=DFS con pesos]
	dfs(u, v, dist + peso);
\end{lstlisting}

La complejidad sigue siendo $\mathcal{O}(n)$ porque el grafo es un árbol.


\section{Orden Topológico}

Se te proporciona un grafo dirigido con $n$ vértices y $m$ aristas. Tienes que encontrar un orden de los vértices, de modo que cada arista vaya desde el vértice con un índice menor hasta un vértice con un índice mayor.

En otras palabras, se trata de encontrar una permutación de los vértices (orden topológico) que corresponda al orden definido por todas las aristas del grafo. Un orden topológico puede no existir en absoluto. Solo existe si el grafo dirigido no contiene ciclos (DAG).

\subsection*{Ejemplo de uso}

Un problema común en el que se produce la ordenación topológica es el siguiente. Hay $n$ variables con valores desconocidos. Para algunas variables, sabemos que una de ellas es menor que la otra. Debes comprobar si estas restricciones son contradictorias y, si no lo son, mostrar las variables en orden ascendente (si hay varias respuestas posibles, mostrar cualquiera de ellas). Es fácil notar que este es exactamente el problema de encontrar el orden topológico de un grafo con $n$ vértices.

\subsection*{¿Cómo funciona con DFS?}

Supongamos que el grafo es acíclico. ¿Qué hace la búsqueda en profundidad (DFS)?

Al comenzar desde algún vértice $v$, DFS intenta recorrer a lo largo de todas las aristas que salen de $v$. Se detiene en los bordes cuyos extremos ya han sido visitados previamente, y recorre el resto de los bordes y continúa recursivamente en sus extremos.

Por lo tanto, en el momento en que la llamada a la función $\text{dfs}(v)$ ha terminado, todos los vértices que son alcanzables desde $v$ han sido visitados por la búsqueda, ya sea directamente (a través de una arista) o indirectamente.

Añadamos el vértice $v$ a una lista cuando terminemos $\text{dfs}(v)$. Como todos los vértices alcanzables ya han sido visitados, ya estarán en la lista cuando agreguemos $v$. Hagamos esto para cada vértice del grafo, con una o varias búsquedas en profundidad. Para cada arista dirigida $v \rightarrow u$ en el gráfico, $u$ aparecerá antes en esta lista que $v$, porque $u$ es accesible desde $v$. 

Entonces, si simplemente etiquetamos los vértices en esta lista con $n-1, n-2, \dots, 1, 0$, hemos encontrado un orden topológico del grafo. En otras palabras, la lista representa el orden topológico inverso, por lo que basta con invertir el arreglo al final.

\subsection*{Código en C++}

Aquí se presenta una implementación que asume que el grafo es acíclico, es decir, que existe el orden topológico deseado:

\begin{lstlisting}[caption=Orden Topológico con DFS]
// n = numero de vertices
int n; 
vector<vector<int>> adj; 
vector<bool> visited;
vector<int> ans;

void dfs(int v) {
        visited[v] = true;
        for (int u : adj[v]) {
                if (!visited[u]) {
                        dfs(u);
                }
        }
        ans.push_back(v);
}

void topological_sort() {
        visited.assign(n, false);
        ans.clear();
        for (int i = 0; i < n; ++i) {
                if (!visited[i]) {
                        dfs(i);
                }
        }
        reverse(ans.begin(), ans.end());
}
\end{lstlisting}

La función principal de la solución es \texttt{topological\_sort}, que inicializa las variables DFS, lanza DFS y recibe la respuesta en el vector \texttt{ans}. 

Vale la pena señalar que cuando el grafo \textbf{no} es acíclico, el resultado de \texttt{topological\_sort} aún sería algo significativo en el sentido de que si un vértice $u$ es alcanzable desde el vértice $v$, pero no al revés, el vértice $v$ siempre aparecerá primero en el array resultante.

\section{Componentes conexas}
\subsection*{Obtener componentes conexas diferentes de un grafo}
\begin{lstlisting}[language=C++, caption={Código para obtener componentes conexas}]
#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> adj;
vector<bool> vis;

void dfs(int u, vector<int>& componente) {
    vis[u] = true;
    componente.push_back(u);
    for (int v : adj[u]) {
        if (!vis[v]) {
            dfs(v, componente);
        }
    }
}

int main() {
    int n, m;
    cin >> n >> m;

    adj.resize(n);
    vis.assign(n, false);

    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        // si los nodos están numerados de 0 a n-1
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    vector<vector<int>> componentes;
    for (int i = 0; i < n; i++) {
        if (!vis[i]) {
            vector<int> componente;
            dfs(i, componente);
            componentes.push_back(componente);
        }
    }

    cout << "Cantidad de componentes: " << componentes.size() << "\n\n";
    for (size_t i = 0; i < componentes.size(); i++) {
        cout << "Componente " << i + 1 << ": ";
        for (int nodo : componentes[i]) {
            cout << nodo << " ";
        }
        cout << "\n";
    }
}
\end{lstlisting}

\section{Matching (Algoritmo Húngaro)}
\subsection*{¿Para qué sirve el algoritmo Húngaro?}
El algoritmo húngaro resuelve el \textit{problema de asignación}: dadas $n$ tareas y $n$ trabajadores (o $n$ filas y $n$ columnas de una matriz de costos), se desea asignar cada tarea a un trabajador de forma que se minimice la suma total de los costos. En términos de grafos, es encontrar un \textit{matching perfecto} de peso mínimo en un grafo bipartito completo.

Este problema aparece frecuentemente en programación competitiva, por ejemplo para:
\begin{itemize}
  \item Asignar trabajos a máquinas minimizando el tiempo total.
  \item Emparejar puntos de dos imágenes minimizando la suma de distancias.
  \item Resolver sub-problemas de coloreado de árboles donde los colores de los hijos se asignan mediante un coste.
\end{itemize}

\subsection*{Cómo funciona el algoritmo}
El algoritmo mantiene dos potenciales $u[i]$ (filas) y $v[j]$ (columnas) con la condición $u[i]+v[j]\leq A[i][j]$. Las aristas donde la igualdad se cumple son rígidas y forman un grafo $H$. Se busca incrementar el tamaño del matching máximo de $H$ usando caminos aumentantes (Kuhn). Si no hay camino aumentante, se reajusta el potencial aumentando $u$ y disminuyendo $v$ en la cantidad mínima $\Delta$ que permite crear nuevas aristas rígidas. Repetimos hasta obtener un matching perfecto. La versión optimizada tiene complejidad $O(n^3)$.

\subsection*{Implementación en C++ (versión compacta)}
\begin{lstlisting}[language=C++,caption=Implementación del algoritmo Húngaro]
#include <bits/stdc++.h>
using namespace std;

#define fore(i,a,b) for(ll i=(a);i<(b);i++)
#define all(x) (x).begin(), (x).end()
#define pb push_back 
#define fst first 
#define snd second 
#define sz(x) ((int)x.size())
#define FIN ios::sync_with_stdio(0);cin.tie(0);cout.tie(0);

using ll = long long; 
typedef vector<double> vi; 

int n, k;
vector<vector<double>> g;
vector<double> mt;
vector<bool> used;

const double INF = 1e9;

pair<double, vector<int>> hungarian(const vector<vector<double>>& cost) {
    int n = cost.size();
    int m = cost[0].size();
    vector<double> u(n + 1, 0), v(m + 1, 0);
    vector<int> p(m + 1, 0), ans(n + 1, 0);
    for (int i = 1; i <= n; ++i) {
        p[0] = i;
        int j0 = 0;
        vector<double> min_v(m + 1, INF);
        vector<int> used(m + 1, 0);
        vector<int> way(m + 1, 0);
        do {
            used[j0] = 1;
            int i0 = p[j0];
            int j1 = 0;
            double delta = INF;
            for (int j = 1; j <= m; ++j) if (!used[j]) {
                double cur = cost[i0-1][j-1] - u[i0] - v[j];
                if (cur < min_v[j]) { min_v[j] = cur; way[j] = j0; }
                if (min_v[j] < delta) { delta = min_v[j]; j1 = j; }
            }
            for (int j = 0; j <= m; ++j) {
                if (used[j]) { u[p[j]] += delta; v[j] -= delta; }
                else min_v[j] -= delta;
            }
            j0 = j1;
        } while (p[j0] != 0);
        do {
            int j1 = way[j0];
            p[j0] = p[j1];
            j0 = j1;
        } while (j0);
    }
    vector<int> matching(n);
    for (int j = 1; j <= m; ++j) if (p[j]) matching[p[j]-1] = j-1;
    double total_cost = -v[0];
    return {total_cost, matching};
}

int main(){
    FIN;
    ll n; cin>>n; 
    g.resize(n, vector<double>(n));
    fore(i,0,n){
        fore(j,0,n){
            ll num; cin>>num; 
            double num2 = log(num); 
            num2 *= -1;
            g[i][j] = num2;
        }
    }
    auto [min_cost, assignment] = hungarian(g);
    vector<pair<int,int>> res;
    for (int i = 0; i < (int)assignment.size(); ++i) {
        res.push_back({assignment[i] + 1, i + 1}); // (columna, fila)
    }
    sort(all(res));
    for (auto [col, fila] : res) {
        cout << fila << " ";
    }
    cout << "\n";
    return 0;
}
\end{lstlisting}

\subsection*{Explicación paso a paso}
\begin{enumerate}
    \item \textbf{Lectura de la matriz de costos}: en el programa se lee una matriz $n\times n$ y se transforma cada valor $c$ en $-\log(c)$, para convertir un problema de \textbf{maximizar producto} en uno de \textbf{minimizar suma}.
    \item \textbf{Llamada a} \texttt{hungarian}: la función devuelve el coste mínimo y el vector \texttt{assignment} donde \texttt{assignment[i]} indica la columna asignada a la fila $i$.
    \item \textbf{Construcción de la salida}: se crea un vector de pares \texttt{(columna, fila)}, se ordena por columna y se imprimen las filas en orden, tal como se suele requerir en concursos.
\end{enumerate}

\subsection*{Cuándo usarlo}
\begin{itemize}
    \item Cuando el número de filas y columnas es pequeño a mediano ($n \leq 500$ suele ser cómodo).
    \item Cuando se necesita la solución exacta del problema de asignación, no una heurística.
    \item En problemas donde el coste se define por distancias euclidianas, log-probabilidades, pesos de aristas, etc.
\end{itemize}

El algoritmo húngaro es una herramienta esencial en la caja de trucos de cualquier competidor de programación competitiva.
